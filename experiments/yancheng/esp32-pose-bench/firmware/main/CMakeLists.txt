if(CONFIG_BENCH_POSE)
    if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/yolo11n_pose.espdl" OR
       NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/model_manifest.json")
        message(FATAL_ERROR "Run tools/prepare_model.py then tools/build.sh pose first.")
    endif()
    file(READ "${CMAKE_CURRENT_LIST_DIR}/model_manifest.json" manifest)
    string(JSON MODEL_TARGET GET "${manifest}" target)
    if(NOT MODEL_TARGET STREQUAL "c")
        message(FATAL_ERROR "Classic ESP32 requires target=c, not an S3/P4 quantized model.")
    endif()
    string(JSON MODEL_SIZE GET "${manifest}" input_size)
    string(JSON MODEL_SHA GET "${manifest}" sha256)
    file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/yolo11n_pose.espdl" actual_sha)
    if(NOT actual_sha STREQUAL MODEL_SHA)
        message(FATAL_ERROR "Model hash differs from its manifest.")
    endif()
    configure_file(model_meta.h.in model_meta.h @ONLY)
endif()
